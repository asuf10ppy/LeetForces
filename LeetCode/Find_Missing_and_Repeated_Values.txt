
    def findMissingAndRepeatedValues(self, grid: List[List[int]]) -> List[int]:
        lis=[]
        out=[]
        for s in grid:
            lis.extend(s)
        for i in lis:
            if lis.count(i)==2:
                out.append(i)
                break
        for i in range(1,len(lis)+1):
            if i not in lis:
                out.append(i)
        return out
